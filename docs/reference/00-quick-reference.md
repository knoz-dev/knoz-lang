
# دليل سريع

مرجع سريع لجميع عناصر بناء الجملة في كنوز.

## المتغيرات

```knoz
دع ثابت = 10;              // ثابت (غير قابل للتغيير)
دع متغير قابل = 20;        // قابل للتغيير
دع مع_نوع: صحيح = 30;     // مع تحديد النوع
```

## الأنواع الأساسية

```knoz
صحيح    // عدد صحيح
ص64     // صحيح 64-بت
عشري    // عدد عشري
ع64     // عشري 64-بت
نص      // سلسلة نصية
منطقي   // صح أو خطأ
حرف     // حرف واحد
```

## العمليات

```knoz
// حسابية
+ - * / %

// مقارنة
== != < > <= >=

// منطقية
و  أو  !
&& ||

// إسناد
= += -= *= /= %=

// نطاق
..    // حصري
..=   // شامل
```

## الدوال

```knoz
// تعريف بسيط
دالة تحية() {
    اطبع("مرحباً");
}

// مع معاملات
دالة جمع(أ: صحيح, ب: صحيح) {
    اطبع_صحيح(أ + ب);
}

// مع قيمة مُرجعة
دالة مربع(س: صحيح) -> صحيح {
    أرجع س * س;
}

// لامدا
دع ضعف = |س: صحيح| -> صحيح { س * 2 };
```

## التحكم في التدفق

```knoz
// شرط
إذا شرط {
    // ...
} وإلا إذا شرط_آخر {
    // ...
} وإلا {
    // ...
}

// حلقة طالما
طالما شرط {
    // ...
}

// حلقة لكل
لكل عنصر في مجموعة {
    // ...
}

// حلقة بنطاق
لكل ت في 0..10 {
    // ...
}

// حلقة لا نهائية
كرر {
    // ...
    إذا شرط_خروج {
        أوقف;
    }
}

// التخطي
تابع;  // تخطي للتكرار التالي
أوقف;   // الخروج من الحلقة
```

## المطابقة

```knoz
طابق قيمة {
    0 => { اطبع("صفر"); }
    1 | 2 | 3 => { اطبع("صغير"); }
    4..=10 => { اطبع("متوسط"); }
    _ => { اطبع("افتراضي"); }
}
```

## المجموعات

```knoz
// مصفوفة (حجم ثابت)
دع أ: [صحيح; 5] = [1, 2, 3, 4, 5];

// صف
دع ص = (10, "نص", صح);
دع أول = ص.0;

// متجه
دع متغير م: متجه<صحيح> = متجه[];
م.ادفع(10);

// قاموس
دع متغير ق: قاموس<نص, صحيح> = قاموس[];
ق.أضف("مفتاح", 100);
```

## الهياكل

```knoz
هيكل اسم_الهيكل {
    حقل١: نوع١,
    حقل٢: نوع٢
}

نفذ اسم_الهيكل {
    دالة جديد() -> ذاتي {
        أرجع ذاتي { حقل١: قيمة, حقل٢: قيمة };
    }
    
    دالة طريقة(&ذاتي) {
        // ...
    }
}
```

## التعدادات

```knoz
تعداد اسم_التعداد {
    قيمة١,
    قيمة٢,
    قيمة٣(نوع)
}

// الاختياري
اختياري<ن>  // بعض(ق) أو عدم

// النتيجة
نتيجة<ن, خ>  // موافق(ق) أو فشل(خ)
```

## السمات

```knoz
سمة اسم_السمة {
    دالة طريقة(&ذاتي);
}

نفذ اسم_السمة لـ اسم_الهيكل {
    دالة طريقة(&ذاتي) {
        // التنفيذ
    }
}
```

## الأنواع العامة

```knoz
// هيكل عام
هيكل صندوق<ن> {
    قيمة: ن
}

// دالة عامة
دالة اطبع<ن>(قيمة: ن) {
    // ...
}

// ثوابت عامة
هيكل مصفوفة<ن, ثابت حجم: صحيح> {
    عناصر: [ن; حجم]
}
```


## التزامن

```knoz
// خيط جديد
دع خ = أطلق || {
    // كود الخيط
};
خ.انضم();

// قفل
دع قفل = قفل_تبادلي(0);
{
    دع م = قفل.أقفل();
    // التعديل الآمن
}

// قناة
دع (مرسل, مستقبل) = قناة();
مرسل.أرسل(قيمة);
دع ق = مستقبل.استقبل();
```

## الملفات

```knoz
// القراءة
دع محتوى = اقرأ_ملف("مسار/الملف");

// الكتابة

// مقبض ملف
دع ملف = افتح_ملف("مسار", "r");
دع سطر = اقرأ_سطر_ملف(ملف);
أغلق_ملف(ملف);
```
