# الخيوط والتزامن

كنوز تدعم البرمجة المتزامنة من خلال الخيوط والقنوات وآليات التزامن المختلفة.

## إطلاق الخيوط

### إطلاق خيط بدالة

```knoz
دالة عمل_فرعي() {
    اطبع("الخيط الفرعي: بدأ التنفيذ")
    لكل ي في 1..4 {
        اطبع("الخيط الفرعي: العمل رقم ")
        اطبع_صحيح(ي)
    }
    اطبع("الخيط الفرعي: انتهى التنفيذ")
}

دالة رئيسية() {
    اطبع("الخيط الرئيسي: إطلاق خيط فرعي...")
    دع مقبض = أطلق(عمل_فرعي())
    
    اطبع("الخيط الرئيسي: يتابع عمله...")
    
    // انتظار انتهاء الخيط
    انضم(مقبض)
    اطبع("الخيط الرئيسي: الخيط الفرعي انتهى!")
}
```

### إطلاق خيط بكتلة مباشرة

```knoz
دع مقبض = أطلق {
    اطبع("خيط الكتلة: مرحباً!")
    اطبع("خيط الكتلة: أنا خيط منفصل")
}

انضم(مقبض)
```

### إطلاق عدة خيوط

```knoz
دالة حساب_مجموع() {
    دع متغير مجموع = 0;
    لكل ي في 1..11 {
        مجموع = مجموع + ي;
    }
    اطبع_صحيح(مجموع);
}

دالة طباعة_أرقام() {
    لكل ي في 1..6 {
        اطبع_صحيح(ي);
    }
}

دالة رئيسية() {
    دع خيط1 = أطلق(حساب_مجموع());
    دع خيط2 = أطلق(طباعة_أرقام());
    
    انضم(خيط1);
    انضم(خيط2);
}
```

## القفل (Mutex)

لحماية البيانات المشتركة:

```knoz
دع قفلي = قفل_جديد()

// قفل
قفل_أقفل(قفلي)
اطبع("البيانات محمية")

// فتح
قفل_افتح(قفلي)

// محاولة القفل (غير مُعيق)
دع نجح = قفل_حاول(قفلي)
إذا نجح {
    // استخدم البيانات
    قفل_افتح(قفلي)
}

// تحرير
قفل_حرر(قفلي)
```

## مجموعة الانتظار (WaitGroup)

للانتظار حتى انتهاء عدة عمليات:

```knoz
دع انتظار = مجموعة_انتظار_جديد()

// إضافة عدد العمليات
مجموعة_انتظار_أضف(انتظار, 2)

// في كل خيط عند الانتهاء
مجموعة_انتظار_تم(انتظار)

// الانتظار حتى انتهاء الكل
مجموعة_انتظار_انتظر(انتظار)

// تحرير
مجموعة_انتظار_حرر(انتظار)
```

## العمليات الذرية

للعمليات الآمنة على المتغيرات المشتركة:

```knoz
دع متغير عدد = 0

// إضافة ذرية
دع ناتج = ذري_أضف(&عدد, 10)
اطبع_صحيح(ناتج)  // 10

دع ناتج2 = ذري_أضف(&عدد, 5)
اطبع_صحيح(ناتج2)  // 15
```

## القنوات (Channels)

للتواصل بين الخيوط:

```knoz
// إنشاء قناة
متغير ق = قناة<صحيح>(3)

// إرسال
أرسل(ق, 10)
أرسل(ق, 20)
أرسل(ق, 30)

// استقبال
دع ق1 = استقبل(ق)  // 10
دع ق2 = استقبل(ق)  // 20

// التحقق من الحالة
إذا قناة_بيانات(ق) {
    اطبع("القناة تحتوي على بيانات")
}

إذا قناة_فارغة(ق) {
    اطبع("القناة فارغة")
}

// إغلاق القناة
أغلق(ق)

إذا قناة_مغلقة(ق) {
    اطبع("القناة مغلقة")
}
```

## المثال الكامل

```knoz
// دالة للخيط الفرعي
دالة عمل_خيط() {
    اطبع("الخيط الفرعي يعمل...")
    لكل ي في 1..4 {
        اطبع_صحيح(ي)
    }
}

دالة رئيسية() -> صحيح {
    اطبع("=== التزامن والخيوط ===")
    
    // === القفل ===
    اطبع("=== القفل ===")
    دع قفلي = قفل_جديد()
    
    قفل_أقفل(قفلي)
    اطبع("القفل مغلق")
    
    قفل_افتح(قفلي)
    اطبع("القفل مفتوح")
    
    قفل_حرر(قفلي)
    
    // === مجموعة الانتظار ===
    اطبع("=== مجموعة الانتظار ===")
    دع انتظار = مجموعة_انتظار_جديد()
    
    مجموعة_انتظار_أضف(انتظار, 2)
    مجموعة_انتظار_تم(انتظار)
    مجموعة_انتظار_تم(انتظار)
    مجموعة_انتظار_انتظر(انتظار)
    
    اطبع("انتهى الانتظار")
    مجموعة_انتظار_حرر(انتظار)
    
    // === العمليات الذرية ===
    اطبع("=== العمليات الذرية ===")
    دع متغير عدد = 0
    
    دع ناتج = ذري_أضف(&عدد, 10)
    اطبع_صحيح(ناتج)
    
    // === القنوات ===
    اطبع("=== القنوات ===")
    متغير ق = قناة<صحيح>(3)
    
    أرسل(ق, 10)
    أرسل(ق, 20)
    
    دع ق1 = استقبل(ق)
    دع ق2 = استقبل(ق)
    
    اطبع_صحيح(ق1)
    اطبع_صحيح(ق2)
    
    أغلق(ق)
    
    // === إطلاق الخيوط ===
    اطبع("=== إطلاق الخيوط ===")
    دع مقبض = أطلق(عمل_خيط())
    اطبع("الخيط الرئيسي يتابع")
    انضم(مقبض)
    اطبع("الخيط الفرعي انتهى")
    
    أرجع 0
}
```
