# السمات (Traits)

السمات تُشبه الواجهات في لغات أخرى. تُحدد سلوكاً مشتركاً يمكن لأنواع مختلفة تنفيذه.

## تعريف سمة

```knoz
سمة قابل_للحساب {
    دالة قيمة(&ذاتي) -> ص64;
}
```

## تنفيذ سمة لهيكل

```knoz
هيكل نقطة {
    س: ص64,
    ص: ص64,
}

نفذ قابل_للحساب لـ نقطة {
    دالة قيمة(&ذاتي) -> ص64 {
        أرجع ذاتي.س + ذاتي.ص;
    }
}
```

## استخدام السمة

```knoz
دع نق = نقطة { س: 10, ص: 20 };
دع الناتج = نق.قيمة();  // 30
اطبع_صحيح(الناتج);
```

## تنفيذ نفس السمة لأنواع مختلفة

```knoz
سمة قابل_للحساب {
    دالة قيمة(&ذاتي) -> ص64;
}

// النقطة
هيكل نقطة {
    س: ص64,
    ص: ص64,
}

نفذ قابل_للحساب لـ نقطة {
    دالة قيمة(&ذاتي) -> ص64 {
        أرجع ذاتي.س + ذاتي.ص;
    }
}

// العداد
هيكل عداد {
    قيمة: ص64,
}

نفذ قابل_للحساب لـ عداد {
    دالة قيمة(&ذاتي) -> ص64 {
        أرجع ذاتي.قيمة * 2;
    }
}
```

## سمات متعددة

يمكن لهيكل تنفيذ عدة سمات:

```knoz
سمة قابل_للطباعة {
    دالة اطبع(&ذاتي);
}

سمة قابل_للمقارنة {
    دالة يساوي(&ذاتي, آخر: &ذاتي) -> منطقي;
}
```

## المثال الكامل

```knoz
// سمة للحساب
سمة قابل_للحساب {
    دالة قيمة(&ذاتي) -> ص64;
}

// هيكل نقطة
هيكل نقطة {
    س: ص64,
    ص: ص64,
}

نفذ قابل_للحساب لـ نقطة {
    دالة قيمة(&ذاتي) -> ص64 {
        أرجع ذاتي.س + ذاتي.ص;
    }
}

// هيكل عداد
هيكل عداد {
    قيمة: ص64,
}

نفذ قابل_للحساب لـ عداد {
    دالة قيمة(&ذاتي) -> ص64 {
        أرجع ذاتي.قيمة * 2;
    }
}

دالة رئيسية() -> صحيح {
    اطبع("=== السمات ===");
    
    // إنشاء نقطة
    دع نق = نقطة { س: 10, ص: 20 };
    اطبع("مجموع النقطة (10+20): ");
    اطبع_صحيح(نق.قيمة());
    
    // إنشاء عداد
    دع عد = عداد { قيمة: 50 };
    اطبع("ضعف العداد (50*2): ");
    اطبع_صحيح(عد.قيمة());
    
    // مجموع القيم
    دع مجموع = نق.قيمة() + عد.قيمة();
    اطبع("المجموع الكلي: ");
    اطبع_صحيح(مجموع);
    
    أرجع 0;
}
```
