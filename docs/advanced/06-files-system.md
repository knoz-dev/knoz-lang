# الملفات والنظام

كنوز توفر دوال للتعامل مع الملفات ونظام التشغيل.

## دوال الملفات

### فتح ملف

```knoz
// أوضاع الفتح:
// "w" - للكتابة (ينشئ ملف جديد)
// "r" - للقراءة
// "a" - للإضافة

دع ملف = افتح_ملف("/tmp/test.txt", "w");

إذا ملف != 0 {
    اطبع("تم فتح الملف بنجاح");
}
```

### كتابة في ملف

```knoz
دع ملف = افتح_ملف("/tmp/test.txt", "w");

إذا ملف != 0 {
    اكتب_نص_ملف("السطر الأول\n", ملف);
    اكتب_نص_ملف("السطر الثاني\n", ملف);
    اكتب_نص_ملف("السطر الثالث\n", ملف);
    
    أغلق_ملف(ملف);
}
```

### الإضافة إلى ملف

```knoz
دع ملف = افتح_ملف("/tmp/test.txt", "a");

إذا ملف != 0 {
    اكتب_نص_ملف("سطر إضافي\n", ملف);
    أغلق_ملف(ملف);
}
```

### قراءة ملف

```knoz
دع ملف = افتح_ملف("/tmp/test.txt", "r");

إذا ملف != 0 {
    دع موقع = موقع_ملف(ملف);
    اطبع_صحيح(موقع);
    أغلق_ملف(ملف);
}
```

### حذف ملف

```knoz
احذف_ملف("/tmp/test.txt");
```

## دوال النظام

### الوقت الحالي

```knoz
دع الوقت = وقت();  // ثواني منذ 1970
اطبع_صحيح(الوقت);
```

### قياس وقت التنفيذ

```knoz
دع بداية = وقت();

// تنفيذ كود يستغرق وقتاً
دع متغير مجموع = 0;
لكل ي في 1..1000000 {
    مجموع = مجموع + ي;
}

دع نهاية = وقت();
دع الفرق = نهاية - بداية;
اطبع("الوقت المستغرق:");
اطبع_صحيح(الفرق);
```

### الأرقام العشوائية

```knoz
دع رقم = عشوائي();
اطبع_صحيح(رقم);

// تحويل لنطاق 1-100
دع متغير رقم_محدود = (رقم % 100) + 1;
إذا رقم_محدود < 0 {
    رقم_محدود = رقم_محدود * -1;
}
```

## إدارة الذاكرة

### حجز الذاكرة

```knoz
دع حجم = 80;  // بالبايت
دع ذاكرة = احجز(حجم);

إذا ذاكرة != 0 {
    اطبع("تم حجز الذاكرة");
}
```

### تحرير الذاكرة

```knoz
حرر(ذاكرة);
اطبع("تم تحرير الذاكرة");
```

:::caution[هام]
دائماً حرر الذاكرة بعد الانتهاء من استخدامها لتجنب تسرب الذاكرة
:::

## المثال الكامل

```knoz
دالة رئيسية() -> صحيح {
    اطبع("=== الملفات والنظام ===");
    
    // === كتابة ملف ===
    اطبع("=== كتابة ملف ===");
    دع ملف = افتح_ملف("/tmp/knoz_test.txt", "w");
    
    إذا ملف != 0 {
        اكتب_نص_ملف("السطر الأول\n", ملف);
        اكتب_نص_ملف("السطر الثاني\n", ملف);
        أغلق_ملف(ملف);
        اطبع("تم كتابة الملف");
    }
    
    // === إضافة للملف ===
    اطبع("=== إضافة للملف ===");
    دع ملف_اضافة = افتح_ملف("/tmp/knoz_test.txt", "a");
    
    إذا ملف_اضافة != 0 {
        اكتب_نص_ملف("سطر إضافي\n", ملف_اضافة);
        أغلق_ملف(ملف_اضافة);
        اطبع("تم إضافة سطر");
    }
    
    // === حذف الملف ===
    احذف_ملف("/tmp/knoz_test.txt");
    اطبع("تم حذف الملف");
    
    // === الوقت ===
    اطبع("=== الوقت ===");
    دع الوقت = وقت();
    اطبع("الوقت الحالي:");
    اطبع_صحيح(الوقت);
    
    // === الذاكرة ===
    اطبع("=== الذاكرة ===");
    دع ذاكرة = احجز(100);
    
    إذا ذاكرة != 0 {
        اطبع("تم حجز 100 بايت");
        حرر(ذاكرة);
        اطبع("تم تحرير الذاكرة");
    }
    
    // === الأرقام العشوائية ===
    اطبع("=== أرقام عشوائية ===");
    لكل ي في 1..4 {
        دع رقم = عشوائي();
        اطبع_صحيح(رقم);
    }
    
    أرجع 0;
}
```
